    UPGRADE_FAIL_LIMIT = 8,
    DRAGON_UPGRADE_FAIL_LIMIT = 3,        -- unexplained upgrade rejections of one dragon (cash confirmed) before it is treated as max level
    TYPE_FAIL_LIMIT = 6,
    MATCH_WAIT = 30,                      -- seconds to wait for ReplicatedStorage.Match before running ungated
    UPGRADE_ARG_MODE = "model",           -- [handoff] only "model" exists in v3
    SPOT_FAIL_TTL = math.huge,            -- seconds a rejected coordinate is skipped when cash was confirmed enough (rest of the round)
    SOFT_FAIL_TTL = 20,                   -- same, when cash or price could not be confirmed (the rejection may have been about cash)
    ANCHOR_FULL_TTL = 40,                 -- seconds an anchor counts as full
    TYPE_OFF_TTL = 45,                    -- first off period of a type after repeated rejected anchors; doubles each time (max 600)
    UPGRADE_BACKOFF = 30,                 -- seconds upgrades pause after UPGRADE_FAIL_LIMIT unexplained failures
    REJECT_BACKOFF = 1.5,                 -- seconds an action kind (place/upgrade of a type) is skipped after a cash-type rejection
    EXPECTED_MAP_HINT = "volcano",        -- recorded spots are for Volcano Fortress; a warning is logged when the live map name lacks this text
    STOP_ON_END = true,                   -- true: stop at Won/Lost. false: wait for the next round and continue
    CASH_TEXT_PATH = nil,                 -- optional "Gui/Child/Label" under PlayerGui
    CASTLE_HP_TEXT_PATH = nil,            -- optional, first integer in the text; location unknown until scanned
    LOG_FILE = "DragonTD_Engine_log.txt",
    STATUS_EVERY = 20,
}

local function path(root, p)
    for part in p:gmatch("[^/]+") do root = root and root:FindFirstChild(part) end
    return root
end

local function exactNumber(v)
    if type(v) == "number" then
        if v == v and v >= 0 and v < math.huge then return v end
        return nil
    end
    local s = tostring(v or ""):gsub("<[^>]+>", ""):gsub(",", ""):gsub("%$", "")
    s = s:match("^%s*(.-)%s*$")
    if not s:match("^%d+%.?%d*$") then return nil end
    return tonumber(s)
end

local function shuffled(list)
    local out = table.clone(list)
    for i = #out, 2, -1 do
        local j = math.random(1, i)
        out[i], out[j] = out[j], out[i]
    end
    return out
end

local function spotKey(pos) return string.format("%d,%d", math.floor(pos.X + 0.5), math.floor(pos.Z + 0.5)) end

local function fmtArg(a)
    if typeof(a) == "Instance" then return "<" .. a.ClassName .. " " .. a:GetFullName() .. ">" end
    return typeof(a) .. ":" .. tostring(a)
end

local function countKeys(t) local n = 0 for _ in pairs(t) do n = n + 1 end return n end

local function createEngine(userCfg, ctx)
    ctx = ctx or {}
    local CONFIG = {}
    for k, v in pairs(DEFAULTS) do CONFIG[k] = v end
    for k, v in pairs(userCfg or {}) do CONFIG[k] = v end

    local E = { Config = CONFIG, Running = false, StatusText = "Idle", Last = "-", Detected = {} }
    local token = 0

    -- per-round state
    local typedSpots, failedSpots, typeFails, typeOff, placeCount, placedLog, upgradeCooldown, anchorState
    local typeOffCount, turn, cursor, turnsDone
    local placeFailRun, upgradeFirst, noSpotUntil, actionBlockedUntil, dragonUpFails, dragonMaxed
    local typeCap = {}                -- per round: type -> number of dragons after which the server refused more
    local roundStats = { placeOk = 0, placeRej = 0, upgOk = 0, upgRej = 0, rej = {} }
    local placeBackoffUntil, upgradeBlockedUntil = 0, 0
    local mapWarned = false
    local seenReasons = {}            -- every distinct server rejection text is logged once
    local learnedBad = {} -- generated candidates the server rejected while cash was sufficient; kept across rounds, keyed by map
    local upgradeFailStreak = 0
    local workingMode = nil
    local warnedEmptyList, warnedUngated = false, false
    local startedAt = 0

    ----------------------------------------------------------------------------------------
    -- logging
    ----------------------------------------------------------------------------------------
    local logLines, lastFlush = {}, 0
    local function flushLog(force)
        if type(writefile) ~= "function" then return end
        if not force and os.clock() - lastFlush < 5 then return end
        lastFlush = os.clock()
        pcall(writefile, CONFIG.LOG_FILE, table.concat(logLines, "\n"))
    end
    local function log(fmt, ...)
        local ok, msg = pcall(string.format, fmt, ...)
        if not ok then msg = tostring(fmt) end
        local line = string.format("[%7.1f] %s", os.clock(), msg)
        print("[DTD] " .. line)
        logLines[#logLines + 1] = line
        while #logLines > 600 do table.remove(logLines, 1) end
        flushLog(false)
    end
    local function setStatus(text)
        if E.StatusText == text then return end
        E.StatusText = text
        if ctx.onStatus then pcall(ctx.onStatus, text) end
    end

    local function resetRound()
        typedSpots = {}
        for name, list in pairs(SPOTS_BY_TYPE) do typedSpots[name] = shuffled(list) end
        failedSpots, typeFails, typeOff, placeCount, placedLog, upgradeCooldown, anchorState = {}, {}, {}, {}, {}, {}, {}
        typeCap, typeOffCount = {}, {}
        turn, cursor, turnsDone = nil, 0, {}
        placeFailRun, upgradeFirst, noSpotUntil, actionBlockedUntil, dragonUpFails, dragonMaxed = {}, {}, {}, {}, {}, {}
        roundStats = { placeOk = 0, placeRej = 0, upgOk = 0, upgRej = 0, rej = {} }
        placeBackoffUntil, upgradeBlockedUntil = 0, 0
        mapWarned = false
        upgradeFailStreak = 0
    end
    resetRound()

    ----------------------------------------------------------------------------------------
    -- rejection handling
    ----------------------------------------------------------------------------------------
    -- The server's real rejection texts are unknown. PERMANENT_PATTERNS are exactly the three the hub itself treats as permanent
    -- ([hub-text]); the other two lists are guesses. Every distinct reason is logged once ("REASON ...") so they can be corrected.
    local PERMANENT_PATTERNS = { "unsupported", "invalid unit", "not equipped" }
    local CASH_PATTERNS = { "cash", "afford", "insufficient", "not enough", "funds" }
    local LIMIT_PATTERNS = { "limit", "too many", "maximum", "max " }

    -- returns class ("error" | "permanent" | "cash" | "limit" | "unknown"), lowercase reason text
    local function classifyReject(ok, r1, r2)
        local reason = r2
        if reason == nil and type(r1) == "string" then reason = r1 end
        local text = tostring(reason or ""):lower()
        if not ok then return "error", text end
        for _, p in ipairs(PERMANENT_PATTERNS) do
            if text:find(p, 1, true) then return "permanent", text end
        end
        for _, p in ipairs(CASH_PATTERNS) do
            if text:find(p, 1, true) then return "cash", text end
        end
        for _, p in ipairs(LIMIT_PATTERNS) do
            if text:find(p, 1, true) then return "limit", text end
        end
        return "unknown", text
    end

    local function noteReason(kind, class, text)
        local key = kind .. "|" .. text
        if seenReasons[key] then return end
        seenReasons[key] = true
        log("REASON %s class=%s text=%q", kind, class, text:sub(1, 160))
    end

    local function spotFailed(key)
        local t = failedSpots[key]
        return t ~= nil and t > os.clock()
    end

    local function typeIsOff(n)
        local t = typeOff[n]
        return t ~= nil and t > os.clock()
    end

    local function anchorIsFull(st)
        return st ~= nil and st.fullUntil ~= nil and st.fullUntil > os.clock()
    end

    ----------------------------------------------------------------------------------------
    -- game readers
    ----------------------------------------------------------------------------------------
    local function playerGui() return LP:FindFirstChildOfClass("PlayerGui") end

    local function getCash()
        local n = exactNumber(LP:GetAttribute("Cash"))
        if n ~= nil then return n end
        local o = path(LP, "leaderstats/Cash")
        if o and o:IsA("ValueBase") then
            local ok, v = pcall(function() return o.Value end)
            if ok then return exactNumber(v) end
        end
        if CONFIG.CASH_TEXT_PATH then
            local lbl = path(playerGui(), CONFIG.CASH_TEXT_PATH)
            if lbl then
                local ok, t = pcall(function() return lbl.Text end)
                if ok then return exactNumber(t) end
            end
        end
        return nil
    end

    local function getCastleHP()
        if not CONFIG.CASTLE_HP_TEXT_PATH then return nil end
        local lbl = path(playerGui(), CONFIG.CASTLE_HP_TEXT_PATH)
        if not lbl then return nil end
        local ok, t = pcall(function() return lbl.Text end)
        local digits = ok and tostring(t):gsub(",", ""):match("%d+")
        return digits and tonumber(digits) or nil
    end

    local function matchObj() return RS:FindFirstChild("Match") end
    local function matchState()
        local m = matchObj()
        return m and m:GetAttribute("State") or nil, m
    end

    -- Live map name. Sources: Match attribute "Map" (the engine log already reads it) or the Workspace attribute "Map" (the hub's
    -- MatchMap reads that one). Which of them exists and its exact format are unverified; "?" when neither is a string.
    local function mapKey()
        local m = matchObj()
        local v = m and m:GetAttribute("Map")
        if type(v) ~= "string" or v == "" then v = WS:GetAttribute("Map") end
        if type(v) == "string" and v ~= "" then return v end
        return "?"
    end

    local function checkMap()
        if mapWarned or not CONFIG.EXPECTED_MAP_HINT then return end
        local mk = mapKey()
        if mk == "?" then return end
        mapWarned = true
        local norm = mk:lower():gsub("[^%a]", "")
        if norm:find(CONFIG.EXPECTED_MAP_HINT, 1, true) then
            log("Map check: %q matches the map the recorded spots were taken on.", mk)
        else
            log("WARNING: live map is %q but SPOTS_BY_TYPE were recorded on Volcano Fortress. Placements will mostly be rejected on another map.", mk)
        end
    end

    -- live game speed from Match.Speed (the hub reads the same attribute); 1 when unreadable
    local function gameSpeed()
        local m = matchObj()
        local sp = m and tonumber(m:GetAttribute("Speed"))
        if sp and sp >= 1 and sp <= 8 then return sp end
        return 1
    end

    local function matchEnded()
        local st = matchState()
        if st == "Won" or st == "Lost" then return true, st end
        local hp = getCastleHP()
        if hp ~= nil and hp <= 0 then return true, "CastleHP=0" end
        return false
    end

    -- returns paused:boolean, reason:string|nil
    local function matchPaused()
        local st, m = matchState()
        if not m then
            if os.clock() - startedAt < CONFIG.MATCH_WAIT then return true, "waiting for Match object" end
            if not warnedUngated then
                warnedUngated = true
                log("WARNING: ReplicatedStorage.Match not found after %ds; running without state gating.", CONFIG.MATCH_WAIT)
            end
            return false
        end
        if st == "Waiting" or st == "RelicPick" then return true, "State=" .. tostring(st) end
        if tostring(m:GetAttribute("RelicPick") or "") ~= "" then return true, "relic pick" end
        return false
    end

    local modules = {}
    local function shared(name)
        if modules[name] then return modules[name] end
        local inst = path(RS, "Shared/" .. name)
        if inst then
            local ok, res = pcall(require, inst)
            if ok and type(res) == "table" then modules[name] = res return res end
        end
        return nil
    end

    local function price(name, level)
        local gc = shared("GameConfig")
        local d = gc and gc.DRAGONS and gc.DRAGONS[name]
        if not d or type(level) ~= "number" or level % 1 ~= 0 or not d.levels or not d.levels[level] then return nil end
        local base = exactNumber(d.levels[level].cost)
        if base == nil then return nil end
        local relics, talents = shared("Relics"), shared("Talents")
        if relics and talents then
            local ok, res = pcall(function()
                return relics.cost(name, level, talents.of(talents.fromPlayerCached(LP), name))
            end)
            if ok then return exactNumber(res) end
            return nil
        end
        return base
    end

    local function maxLevelFor(name)
        local gc = shared("GameConfig")
        local def = gc and gc.DRAGONS and gc.DRAGONS[name]
        if type(def) == "table" and type(def.levels) == "table" and #def.levels > 0 then return #def.levels end
        return CONFIG.MAX_LEVEL_BY_TYPE[name] or CONFIG.MAX_LEVEL
    end

    ----------------------------------------------------------------------------------------
    -- Hotbar detection: slot -> dragon name. The slot GUI has no name [handoff], so match what it does show
    -- (Icon image id, Cost text) against Shared.GameConfig.DRAGONS. Unverified against the real GameConfig layout.
    ----------------------------------------------------------------------------------------
    local function safeGet(obj, prop)
        if not obj then return nil end
        local ok, v = pcall(function() return obj[prop] end)
        return ok and v or nil
    end

    local function idDigits(v)
        return tostring(v or ""):match("(%d%d%d%d%d+)")
    end

    local loggedFields = false
    function E.DetectHotbar()
        local pg = playerGui()
        local bar = pg and path(pg, "DragonHotbar/Bar")
        if not bar then return nil, "DragonHotbar.Bar not found (are you inside a match?)" end
        local gc = shared("GameConfig")
        if not (gc and type(gc.DRAGONS) == "table") then return nil, "Shared.GameConfig.DRAGONS not readable" end

        if not loggedFields then
            loggedFields = true
            for name, def in pairs(gc.DRAGONS) do
                if type(def) == "table" then
                    local keys = {}
                    for k in pairs(def) do keys[#keys + 1] = tostring(k) end
                    table.sort(keys)
                    log("GameConfig.DRAGONS[%s] fields: %s", tostring(name), table.concat(keys, ","))
                    break
                end
            end
        end

        local priceCache = {}
        local function p1(name)
            if priceCache[name] == nil then priceCache[name] = price(name, 1) or false end
            return priceCache[name] or nil
        end

        local names, seen, ambiguous = {}, {}, 0
        for i = 1, 5 do
            local slot = bar:FindFirstChild("Slot" .. i)
            if slot then
                local cost = exactNumber(safeGet(slot:FindFirstChild("Cost"), "Text"))
                local image = safeGet(slot:FindFirstChild("Icon"), "Image")
                local id = idDigits(image)
                local byIcon, byCost = {}, {}
                for name, def in pairs(gc.DRAGONS) do
                    if type(name) == "string" and type(def) == "table" then
                        if cost then
                            local p = p1(name)
                            if p and math.abs(p - cost) < 0.5 then byCost[#byCost + 1] = name end
                        end
                        if id then
                            for _, v in pairs(def) do
                                if (type(v) == "string" or type(v) == "number") and idDigits(v) == id then
                                    byIcon[#byIcon + 1] = name
                                    break
                                end
                            end
                        end
                    end
                end
                table.sort(byIcon) table.sort(byCost)

                local pick, how
                if #byIcon == 1 then pick, how = byIcon[1], "icon"
                elseif #byCost == 1 then pick, how = byCost[1], "cost"
                elseif #byIcon > 1 and #byCost > 0 then
                    local both = {}
                    for _, a in ipairs(byIcon) do
                        for _, b in ipairs(byCost) do if a == b then both[#both + 1] = a end end
                    end
                    if #both == 1 then pick, how = both[1], "icon+cost" end
                end

                if pick then
                    log("HOTBAR Slot%d cost=%s icon=%s -> %s (by %s)", i, tostring(cost), tostring(image), pick, how)
                    if not seen[pick] then seen[pick] = true names[#names + 1] = pick end
                else
                    ambiguous = ambiguous + 1
                    local cand = #byIcon > 0 and byIcon or byCost
                    log("HOTBAR Slot%d cost=%s icon=%s -> AMBIGUOUS/unknown, candidates: %s", i, tostring(cost), tostring(image),
                        #cand > 0 and table.concat(cand, ",") or "none")
                end
            end
        end
        E.Detected = names
        return names, { ambiguous = ambiguous }
    end

    ----------------------------------------------------------------------------------------
    -- dragons
    ----------------------------------------------------------------------------------------
    local function dragonsFolder() return WS:FindFirstChild("Dragons") end

    local function isMine(d)
        return typeof(d) == "Instance" and d:IsA("Model") and d.Parent ~= nil and d.Parent == dragonsFolder()
            and tonumber(d:GetAttribute("Owner")) == LP.UserId
    end

    local function nameOf(d)
        local id = d:GetAttribute("DragonId")
        if type(id) == "string" and id ~= "" then return id end
        return d.Name
    end

    local function levelOf(d) return tonumber(d:GetAttribute("Level")) end

    local function dragonPos(d)
        local s = d:GetAttribute("Spot")
        if typeof(s) == "Vector3" then return s end
        local ok, cf = pcall(function() return d:GetPivot() end)
        return ok and cf.Position or nil
    end

    local function getMyDragons()
        local out, f = {}, dragonsFolder()
        if f then
            for _, d in ipairs(f:GetChildren()) do
                if isMine(d) then out[#out + 1] = d end
            end
        end
        return out
    end

    local function snapshot()
        local set = {}
        for _, d in ipairs(getMyDragons()) do set[d] = true end
        return set
    end

    local function ownedCounts()
        local c = {}
        for _, d in ipairs(getMyDragons()) do
            local n = nameOf(d)
            c[n] = (c[n] or 0) + 1
        end
        return c
    end

    local function activeNames()
        if #CONFIG.DRAGON_LIST > 0 then return CONFIG.DRAGON_LIST end
        if #E.Detected > 0 then return E.Detected end
        local seen, out = {}, {}
        for _, d in ipairs(getMyDragons()) do
            local n = nameOf(d)
            if not seen[n] then seen[n] = true out[#out + 1] = n end
        end
        return out
    end

    local function callRemote(name, ...)
        local r = path(RS, "Remotes/" .. name)
        if not r then return false, "Remote unavailable: " .. name end
        if r:IsA("RemoteFunction") then return pcall(r.InvokeServer, r, ...)
        elseif r:IsA("RemoteEvent") then return pcall(r.FireServer, r, ...) end
        return false, "Unsupported remote class: " .. r.ClassName
    end

    local function nearestRecord(name, pos)
        for _, r in ipairs(placedLog) do
            if r.name == name and (r.pos - pos).Magnitude < 2 then return r end
        end
        return nil
    end

    local function waitForDragon(pred, timeout, myToken)
        local deadline = os.clock() + timeout
        repeat
            for _, d in ipairs(getMyDragons()) do
                if pred(d) then return d end
            end
            task.wait(0.05)
        until os.clock() > deadline or token ~= myToken
        return nil
    end

    local function jitter(myToken)
        local lo, hi = CONFIG.JITTER_MIN, math.max(CONFIG.JITTER_MIN, CONFIG.JITTER_MAX)
        local pause = lo + math.random() * (hi - lo)
        if CONFIG.JITTER_SCALE_BY_SPEED then pause = pause / gameSpeed() end
        task.wait(pause)
        return token == myToken
    end

    ----------------------------------------------------------------------------------------
    -- tryPlace / tryUpgrade: return true when a call (or wait) was made this tick
    ----------------------------------------------------------------------------------------
    local function stillGo(myToken)
        return token == myToken and not matchEnded() and not matchPaused() and not (ctx.blocked and ctx.blocked())
    end

    -- positions of every dragon on the map (any owner), fetched once per decision
    local function occupied()
        local out, f = {}, dragonsFolder()
        if f then
            for _, d in ipairs(f:GetChildren()) do
                if d:IsA("Model") then
                    local p = dragonPos(d)
                    if p then out[#out + 1] = p end
                end
            end
        end
        return out
    end

    local function isTaken(occ, pos)
        for _, p in ipairs(occ) do
            if (p - pos).Magnitude < CONFIG.SPOT_CLEAR_RADIUS then return true end
        end
        return false
    end

    local offsetCache
    local function clusterOffsets()
        local step, rings = CONFIG.CLUSTER_STEP, CONFIG.CLUSTER_RINGS
        local key = tostring(step) .. ":" .. tostring(rings)
        if offsetCache and offsetCache.key == key then return offsetCache.list end
        local list = {}
        for ring = 1, rings do
            local r = ring * step
            local n = math.max(6, math.floor(2 * math.pi * r / step + 0.5))
            local ringPts = {}
            for k = 0, n - 1 do
                local ang = 2 * math.pi * k / n + ring * 0.5
                ringPts[#ringPts + 1] = { key = (k * 0.6180339887) % 1, off = Vector3.new(math.cos(ang) * r, 0, math.sin(ang) * r) }
            end
            -- spread consecutive attempts around the circle so one blocked sector does not use up the failure budget
            table.sort(ringPts, function(a, b) return a.key < b.key end)
            for _, pt in ipairs(ringPts) do list[#list + 1] = pt.off end
        end
        offsetCache = { key = key, list = list }
        return list
    end

    ----------------------------------------------------------------------------------------
    -- spot picking. Anchors come only from SPOTS_BY_TYPE[name]. Pass 1 = every recorded coordinate; pass 2 = rings around
    -- the anchors, used only when no recorded coordinate is free (taken by any account, rejected, or marked full).
    -- returns pos, kind ("anchor" = recorded coordinate, "gen" = generated candidate), anchorKey
    ----------------------------------------------------------------------------------------
    local function pickSpot(name, occ)
        occ = occ or occupied()
        local mk = mapKey()
        local anchors = typedSpots[name] or {}
        for _, anchor in ipairs(anchors) do
            local ak = spotKey(anchor)
            if not anchorIsFull(anchorState[ak]) and not spotFailed(ak) and not isTaken(occ, anchor) then
                return anchor, "anchor", ak
            end
        end
        local offsets = CONFIG.CLUSTER_RINGS > 0 and clusterOffsets() or {}
        if #offsets == 0 then return nil end
        for _, anchor in ipairs(anchors) do
            local ak = spotKey(anchor)
            if not anchorIsFull(anchorState[ak]) then
                for _, off in ipairs(offsets) do
                    local pos = anchor + off
                    local key = spotKey(pos)
                    if not spotFailed(key) and not learnedBad[mk .. ":" .. key] and not isTaken(occ, pos) then
                        return pos, "gen", ak
                    end
                end
            end
        end
        return nil
    end

    -- a type may take another dragon while it is not off, below the given limit and below the cap the server taught us
    local function typeRoom(n, owned, limit)
        local have = owned[n] or 0
        return not typeIsOff(n) and have < limit and have < (typeCap[n] or math.huge)
    end

    local function totalRoom() return #getMyDragons() < CONFIG.MAX_TOTAL end

    local function placeRank(name, pos)
        local rec = nearestRecord(name, pos)
        return rec and rec.idx or 0
    end

    ----------------------------------------------------------------------------------------
    -- place / upgrade executors. Outcomes:
    --   place:   "ok" | "cash" | "rejected" | "nospot" | "taken" | "off" | "limit" | "abort"
    --   upgrade: "ok" | "cash" | "rejected" | "stale" | "abort"
    ----------------------------------------------------------------------------------------
    local function tryPlace(myToken, name)
        local p, cash = price(name, 1), getCash()
        local pos, kind, ak = pickSpot(name)
        if not pos then
            noSpotUntil[name] = os.clock() + CONFIG.NOSPOT_RETRY
            if os.clock() - (E._lastNoSpot or -99) > 10 then
                E._lastNoSpot = os.clock()
                log("No free spot for %s (failed=%d, learned-bad=%d).", name, countKeys(failedSpots), countKeys(learnedBad))
            end
            return "nospot"
        end
        local key = spotKey(pos)
        local retries = kind == "anchor" and CONFIG.PLACE_RETRIES or 1
        local confirmT = kind == "anchor" and CONFIG.CONFIRM_TIMEOUT or CONFIG.GEN_CONFIRM_TIMEOUT

        local lastOk, lastR1, lastR2 = true, false, nil
        for attempt = 1, retries do
            if not jitter(myToken) then return "abort" end
            if not stillGo(myToken) then return "abort" end
            -- another account may have taken this coordinate during the pause: pick again, no penalty
            if isTaken(occupied(), pos) then return "taken" end
            local before = snapshot()
            local cash0, t0 = getCash(), os.clock()
            local ok, r1, r2 = callRemote("PlaceDragon", name, pos)
            local seen
            if not (ok and r1 == false) then
                seen = waitForDragon(function(d)
                    if before[d] or nameOf(d) ~= name then return false end
                    local dp = dragonPos(d)
                    return dp ~= nil and (dp - pos).Magnitude < 2
                end, confirmT, myToken)
            end
            log("PLACE %s %s@%s (anchor %s) try %d/%d: call_ok=%s ret=%s,%s seen=%s dt=%.2fs cash %s -> %s",
                name, kind, key, ak, attempt, retries, tostring(ok), tostring(r1), tostring(r2),
                tostring(seen ~= nil), os.clock() - t0, tostring(cash0), tostring(getCash()))
            if seen or (ok and r1 == true) then
                placeCount[name] = (placeCount[name] or 0) + 1
                placedLog[#placedLog + 1] = { name = name, idx = placeCount[name], pos = pos }
                typeFails[name] = 0
                anchorState[ak] = { fails = 0 }
                roundStats.placeOk = roundStats.placeOk + 1
                E.Last = "Placed " .. name
                return "ok"
            end
            lastOk, lastR1, lastR2 = ok, r1, r2
            local class0, text0 = classifyReject(ok, r1, r2)
            noteReason("place", class0, text0)
            -- a recognised reason will not change by retrying the same coordinate
            if class0 ~= "unknown" and class0 ~= "error" then break end
        end

        roundStats.placeRej = roundStats.placeRej + 1
        local class, text = classifyReject(lastOk, lastR1, lastR2)
        roundStats.rej[class] = (roundStats.rej[class] or 0) + 1
        E.Last = "Place rejected " .. name
        local have = ownedCounts()[name] or 0
        if class == "cash" then
            -- price() disagreed with the server or cash moved; the coordinate itself is fine
            actionBlockedUntil["P:" .. name] = os.clock() + CONFIG.REJECT_BACKOFF
            return "cash"
        elseif class == "permanent" then
            typeOff[name] = math.huge
            log("Type %s switched off for this round: %s", name, text)
            return "off"
        elseif class == "limit" and have > 0 then
            typeCap[name] = have
            log("Type %s capped at %d for this round (server said: %s)", name, have, text)
            return "limit"
        end

        -- unknown reason or thrown error: a coordinate problem. Skipped for the round when cash was confirmed enough, otherwise briefly.
        local paid = p ~= nil and cash ~= nil and cash >= p
        failedSpots[key] = os.clock() + (paid and CONFIG.SPOT_FAIL_TTL or CONFIG.SOFT_FAIL_TTL)
        if kind == "gen" and paid then learnedBad[mapKey() .. ":" .. key] = true end
        local st = anchorState[ak] or { fails = 0 }
        anchorState[ak] = st
        st.fails = st.fails + 1
        if st.fails >= CONFIG.CLUSTER_FAIL_LIMIT then
            st.fullUntil = os.clock() + CONFIG.ANCHOR_FULL_TTL
            st.fails = 0
            log("Anchor %s treated as full for %ds after %d consecutive rejections; moving to the next coordinate.",
                ak, CONFIG.ANCHOR_FULL_TTL, CONFIG.CLUSTER_FAIL_LIMIT)
        end
        if kind == "anchor" then
            typeFails[name] = (typeFails[name] or 0) + 1
            log("Anchor %s rejected for %s (type fails %d/%d)", key, name, typeFails[name], CONFIG.TYPE_FAIL_LIMIT)
            if typeFails[name] >= CONFIG.TYPE_FAIL_LIMIT then
                typeOffCount[name] = (typeOffCount[name] or 0) + 1
                local ttl = math.min(CONFIG.TYPE_OFF_TTL * 2 ^ (typeOffCount[name] - 1), 600)
                typeOff[name] = os.clock() + ttl
                typeFails[name] = 0
                log("Type %s switched off for %ds after %d consecutive rejected anchors. Is it on your hotbar?",
                    name, ttl, CONFIG.TYPE_FAIL_LIMIT)
            end
        end
        return "rejected"
    end

    -- target = the dragon Model (UpgradeDragon takes the Model [handoff]); price p = cost of its next level or nil
    local function tryUpgrade(myToken, target, name, p)
        local lvl, pos = levelOf(target), dragonPos(target)
        if not (lvl and pos) then return "stale" end
        if not jitter(myToken) then return "abort" end
        if not stillGo(myToken) then return "abort" end
        if not isMine(target) or levelOf(target) ~= lvl then return "stale" end -- Model replaced during the wait

        local key = spotKey(pos)
        local cash0, t0 = getCash(), os.clock()
        local ok, r1, r2 = callRemote("UpgradeDragon", target)
        local newD
        if not (ok and r1 == false) then -- an explicit rejection needs no confirmation wait
            newD = waitForDragon(function(d)
                if nameOf(d) ~= name or levelOf(d) ~= lvl + 1 then return false end
                local dp = dragonPos(d)
                return dp ~= nil and (dp - pos).Magnitude < 2
            end, CONFIG.CONFIRM_TIMEOUT, myToken)
        end
        log("UPGRADE %s L%d->L%d mode=model args=[%s] call_ok=%s ret=%s,%s confirmed=%s dt=%.2fs cash %s -> %s",
            name, lvl, lvl + 1, fmtArg(target), tostring(ok), tostring(r1), tostring(r2),
            tostring(newD ~= nil), os.clock() - t0, tostring(cash0), tostring(getCash()))
        if newD then
            workingMode = "model"
            upgradeFailStreak = 0
            dragonUpFails[key] = nil
            roundStats.upgOk = roundStats.upgOk + 1
            E.Last = string.format("Upgraded %s to L%d", name, lvl + 1)
            return "ok"
        end

        roundStats.upgRej = roundStats.upgRej + 1
        local class, text = classifyReject(ok, r1, r2)
        noteReason("upgrade", class, text)
        roundStats.rej[class] = (roundStats.rej[class] or 0) + 1
        E.Last = "Upgrade unconfirmed " .. name
        if class == "cash" then
            actionBlockedUntil["U:" .. name] = os.clock() + CONFIG.REJECT_BACKOFF
            return "cash"
        elseif class == "limit" then
            upgradeCooldown[key] = os.clock() + 120
            log("UPGRADE %s L%d: server reports a limit (%s); skipping this dragon for 120s.", name, lvl, text)
            return "rejected"
        end

        upgradeCooldown[key] = os.clock() + CONFIG.UPGRADE_COOLDOWN
        dragonUpFails[key] = (dragonUpFails[key] or 0) + 1
        local paid = p ~= nil and cash0 ~= nil and cash0 >= p
        if (paid and dragonUpFails[key] >= CONFIG.DRAGON_UPGRADE_FAIL_LIMIT) or dragonUpFails[key] >= 2 * CONFIG.DRAGON_UPGRADE_FAIL_LIMIT then
            dragonMaxed[key] = true
            log("UPGRADE %s L%d: %d unexplained rejections; treating this dragon as max level for this round.", name, lvl, dragonUpFails[key])
        end
        upgradeFailStreak = upgradeFailStreak + 1
        if upgradeFailStreak >= CONFIG.UPGRADE_FAIL_LIMIT then
            upgradeBlockedUntil = os.clock() + CONFIG.UPGRADE_BACKOFF
            upgradeFailStreak = 0
            log("UPGRADES PAUSED for %ds after %d unexplained failures. If this repeats, check the UPGRADE and REASON lines above.",
                CONFIG.UPGRADE_BACKOFF, CONFIG.UPGRADE_FAIL_LIMIT)
        end
        return "rejected"
    end

    ----------------------------------------------------------------------------------------
    -- v3 scheduler: turns per type, no saving
    ----------------------------------------------------------------------------------------
    -- lowest-level upgradable dragon of one type (ties: placed first); returns the Model and the price of its next level (may be nil)
    local function upgradeTargetFor(name)
        local now, list = os.clock(), {}
        for _, d in ipairs(getMyDragons()) do
            if nameOf(d) == name then
                local lvl, pos = levelOf(d), dragonPos(d)
                if lvl and pos then
                    local k = spotKey(pos)
                    if lvl < maxLevelFor(name) and not dragonMaxed[k] and (upgradeCooldown[k] or 0) <= now then
                        list[#list + 1] = { d = d, lvl = lvl, rank = placeRank(name, pos), born = tonumber(d:GetAttribute("Born")) or 0, key = k }
                    end
                end
            end
        end
        if #list == 0 then return nil end
        table.sort(list, function(a, b)
            if a.lvl ~= b.lvl then return a.lvl < b.lvl end
            if a.rank ~= b.rank then return a.rank < b.rank end
            if a.born ~= b.born then return a.born < b.born end
            return a.key < b.key
        end)
        return list[1].d, price(name, list[1].lvl + 1)
    end

    -- types of TYPE_PRIORITY that this account lists (hotbar / DRAGON_LIST / owned), in priority order
    local function listedTypes()
        local have = {}
        for _, n in ipairs(activeNames()) do have[n] = true end
        local out = {}
        for _, n in ipairs(CONFIG.TYPE_PRIORITY) do
            if have[n] then out[#out + 1] = n end
        end
        return out
    end

    -- true while the type can still place another dragon or has a dragon below max level
    local function typeHasWork(name, owned)
        if typeIsOff(name) then return false end
        if typeRoom(name, owned, CONFIG.MAX_PER_TYPE) and totalRoom() then return true end
        for _, d in ipairs(getMyDragons()) do
            if nameOf(d) == name then
                local lvl, pos = levelOf(d), dragonPos(d)
                if lvl and pos and lvl < maxLevelFor(name) and not dragonMaxed[spotKey(pos)] then return true end
            end
        end
        return false
    end

    -- Prism/Titan open when Void or Minigun has GATE_MIN_PLACED placed. Also open when the base types are not listed or
    -- have nothing left to do, so a capped or finished base type does not stall everything.
    local function gateOpen(owned)
        for _, b in ipairs(CONFIG.BASE_TYPES) do
            if (owned[b] or 0) >= CONFIG.GATE_MIN_PLACED then return true end
        end
        local listed = listedTypes()
        for _, b in ipairs(CONFIG.BASE_TYPES) do
            if table.find(listed, b) and typeHasWork(b, owned) then return false end
        end
        return true
    end

    local function eligibleTypes(owned)
        local open = gateOpen(owned)
        local out = {}
        for _, n in ipairs(listedTypes()) do
            local gated = table.find(CONFIG.BASE_TYPES, n) == nil
            if (open or not gated) and typeHasWork(n, owned) then out[#out + 1] = n end
        end
        return out
    end

    local function finishTurn()
        if not turn then return end
        turnsDone[turn.type] = (turnsDone[turn.type] or 0) + 1
        cursor = turn.idx
        if turn.announced then log("TURN end %s", turn.type) end
        turn = nil
    end

    local function advanceStep()
        if not turn then return end
        turn.step = turn.step + 1
        turn.stepFails = 0
        turn.fbLogged = false
        if turn.step > #turn.script then finishTurn() end
    end

    -- next eligible type after the one served last, walking TYPE_PRIORITY in a circle
    local function startTurn(owned)
        local elig = eligibleTypes(owned)
        if #elig == 0 then return false end
        local order = CONFIG.TYPE_PRIORITY
        local pick
        for off = 1, #order do
            local idx = (cursor + off - 1) % #order + 1
            if table.find(elig, order[idx]) then pick = idx break end
        end
        if not pick then return false end
        local name = order[pick]
        local script = (owned[name] or 0) == 0 and CONFIG.FIRST_TURN or CONFIG.NEXT_TURN
        turn = { type = name, idx = pick, script = script, step = 1, stepFails = 0 }
        return true
    end

    local function turnDesc()
        if not turn then return "-" end
        return string.format("%s %s step %d/%d", turn.type, table.concat(turn.script), math.min(turn.step, #turn.script), #turn.script)
    end

    -- nil when the action cannot be done at all right now
    local function buildAction(kind, name, owned)
        if kind == "P" then
            if not typeRoom(name, owned, CONFIG.MAX_PER_TYPE) or not totalRoom() then return nil end
            if (noSpotUntil[name] or 0) > os.clock() then return nil end
            return { kind = "P", name = name, price = price(name, 1) }
        end
        local target, p = upgradeTargetFor(name)
        if not target then return nil end
        return { kind = "U", name = name, target = target, price = p }
    end

    local function affordable(a, cash)
        return a.price == nil or cash == nil or cash >= a.price
    end

    local function isBlocked(a)
        if (actionBlockedUntil[a.kind .. ":" .. a.name] or 0) > os.clock() then return true end
        return a.kind == "U" and os.clock() < upgradeBlockedUntil
    end

    local function describe(a, cash)
        return string.format("%s %s%s costs %s, cash %s", a.kind == "P" and "place" or "upgrade", a.name,
            a.kind == "U" and (" L" .. tostring(levelOf(a.target) or "?")) or "", tostring(a.price or "?"), tostring(cash or "?"))
    end

    -- returns an action, or nil plus a status text
    local function chooseAction()
        local owned = ownedCounts()
        local cash = getCash()

        local primary
        for _ = 1, 12 do
            if not turn and not startTurn(owned) then break end
            local kind = turn.script[turn.step]
            local a = kind and buildAction(kind, turn.type, owned)
            if a then primary = a break end
            if kind then advanceStep() else finishTurn() end -- impossible step (no room, nothing to upgrade): skip it
        end

        -- a place of this type was just rejected: upgrade something first, then search for a spot again
        if primary and primary.kind == "P" and upgradeFirst[primary.name] then
            local up = buildAction("U", primary.name, owned)
            if not (up and affordable(up, cash) and not isBlocked(up)) then
                up = nil
                for _, n in ipairs(eligibleTypes(owned)) do
                    local c = buildAction("U", n, owned)
                    if c and affordable(c, cash) and not isBlocked(c) then up = c break end
                end
            end
            if up then
                up.clearsFirst = primary.name
                return up
            end
            upgradeFirst[primary.name] = nil -- nothing upgradable or affordable right now: go straight back to the spot search
        end

        if primary and affordable(primary, cash) and not isBlocked(primary) then
            primary.primary = true
            return primary
        end

        -- never save: spend the cash on the first affordable action in type priority order (upgrade of the type's lowest dragon first, then a place)
        for _, n in ipairs(eligibleTypes(owned)) do
            for _, kind in ipairs({ "U", "P" }) do
                if not (kind == "P" and upgradeFirst[n]) then
                    local a = buildAction(kind, n, owned)
                    if a and affordable(a, cash) and not isBlocked(a) then
                        if turn and not turn.fbLogged then
                            turn.fbLogged = true
                            log("FALLBACK (no saving): step %s of turn %s not payable now%s -> %s %s",
                                tostring(turn.script[turn.step]), turn.type, primary and (" (" .. describe(primary, cash) .. ")") or "", kind, n)
                        end
                        return a
                    end
                end
            end
        end
        if primary then return nil, "Waiting for cash: " .. describe(primary, cash) end
        return nil, "Nothing to do right now (all placed / maxed, or no free spot)"
    end

    local function noteResult(act, res)
        if res == "abort" or res == "stale" then return end
        local name = act.name
        if act.kind == "P" then
            if res == "ok" then
                placeFailRun[name] = 0
            elseif res == "rejected" then
                placeFailRun[name] = (placeFailRun[name] or 0) + 1
                if placeFailRun[name] <= CONFIG.PLACE_FAIL_UPGRADE_FIRST then upgradeFirst[name] = true end
                if turn and act.primary then turn.stepFails = turn.stepFails + 1 end
            end
        elseif act.clearsFirst and res ~= "cash" then
            upgradeFirst[act.clearsFirst] = nil
        end

        if act.primary and turn then
            if act.kind == "U" then
                if res ~= "cash" then advanceStep() end
            elseif res == "ok" then
                advanceStep()
            elseif res == "rejected" and turn.stepFails >= CONFIG.PLACE_STEP_GIVEUP then
                log("TURN %s: place step abandoned after %d rejections", turn.type, turn.stepFails)
                advanceStep()
            end
        end
    end

    local function ensureNames()
        local names = activeNames()
        if #names == 0 and os.clock() - (E._lastDetect or -99) > 5 then
            E._lastDetect = os.clock()
            pcall(E.DetectHotbar)
            names = activeNames()
        end
        if #names == 0 then
            if not warnedEmptyList then
                warnedEmptyList = true
                log("Dragon list is empty, hotbar detection found nothing and you own no dragons yet: set the list or place one dragon manually.")
            end
            setStatus("Set the dragon list (or place 1 dragon manually)")
        end
        return names
    end

    -- returns true when an action (or its wait) was made this tick
    local function schedulerStep(myToken)
        if #ensureNames() == 0 then return false end
        local act, idle = chooseAction()
        if not act then
            if idle then setStatus(idle) end
            return false
        end
        if turn and not turn.announced then
            turn.announced = true
            log("TURN start %s (turn #%d, owned %d, script %s)", turn.type, (turnsDone[turn.type] or 0) + 1,
                ownedCounts()[turn.type] or 0, table.concat(turn.script))
        end
        local res
        if act.kind == "P" then
            res = tryPlace(myToken, act.name)
        else
            res = tryUpgrade(myToken, act.target, act.name, act.price)
        end
        noteResult(act, res)
        if res == "nospot" or res == "taken" or res == "cash" or res == "off" or res == "limit" then task.wait(0.1) end
        return true
    end

    -- Event-driven idle: wakes on a Cash change, a dragon appearing/disappearing, or a Match state change (IDLE_WAIT is only a safety net).
    local function waitForEvent(timeout, myToken)
        local fired = false
        local conns = {}
        local function hit() fired = true end
        local function hook(sig)
            local ok, c = pcall(function() return sig:Connect(hit) end)
            if ok and c then conns[#conns + 1] = c end
        end
        hook(LP:GetAttributeChangedSignal("Cash"))
        local cashObj = path(LP, "leaderstats/Cash")
        if cashObj and cashObj:IsA("ValueBase") then hook(cashObj.Changed) end
        local f = dragonsFolder()
        if f then
            hook(f.ChildAdded)
            hook(f.ChildRemoved)
        end
        local m = matchObj()
        if m then hook(m:GetAttributeChangedSignal("State")) end
        task.wait(0.1)
        local deadline = os.clock() + timeout
        while not fired and os.clock() < deadline and token == myToken do task.wait(0.05) end
        for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    end

    local function logRoundEnd(why)
        local st, m = matchState()
        local parts = {}
        for _, d in ipairs(getMyDragons()) do
            parts[#parts + 1] = string.format("%s L%s", nameOf(d), tostring(levelOf(d) or "?"))
        end
        table.sort(parts)
        local rej = {}
        for k, v in pairs(roundStats.rej) do rej[#rej + 1] = k .. "=" .. v end
        table.sort(rej)
        log("ROUND END %s | state=%s wave=%s cash=%s map=%s | place ok/rej %d/%d | upgrade ok/rej %d/%d | reject classes [%s] | field: %s",
            tostring(why), tostring(st), tostring(m and m:GetAttribute("Wave")), tostring(getCash()), mapKey(),
            roundStats.placeOk, roundStats.placeRej, roundStats.upgOk, roundStats.upgRej,
            table.concat(rej, ","), #parts > 0 and table.concat(parts, ", ") or "none")
        flushLog(true)
    end

    ----------------------------------------------------------------------------------------
    -- main loop
    ----------------------------------------------------------------------------------------
    local function loop(myToken)
        if not game:IsLoaded() then game.Loaded:Wait() end
        local m = matchObj()
        log("Engine start. user=%s place=%s Match=%s State=%s Map=%s Wave=%s",
            LP.Name, tostring(game.PlaceId), tostring(m ~= nil), tostring(m and m:GetAttribute("State")),
            tostring(m and m:GetAttribute("Map")), tostring(m and m:GetAttribute("Wave")))
        log("Scheduler: priority=[%s] base=[%s] gate>=%d first=%s next=%s | list in use: [%s] | MAX_PER_TYPE=%d MAX_TOTAL=%d",
            table.concat(CONFIG.TYPE_PRIORITY, ","), table.concat(CONFIG.BASE_TYPES, ","), CONFIG.GATE_MIN_PLACED,
            table.concat(CONFIG.FIRST_TURN), table.concat(CONFIG.NEXT_TURN), table.concat(activeNames(), ", "), CONFIG.MAX_PER_TYPE, CONFIG.MAX_TOTAL)
        for _, n in ipairs(activeNames()) do
            if not table.find(CONFIG.TYPE_PRIORITY, n) then log("Type %s is in the list but not in TYPE_PRIORITY: ignored.", n) end
        end
        if getCash() == nil then log("WARNING: cash unreadable (no Player attribute Cash, no leaderstats.Cash); affordability checks disabled.") end
        if not shared("GameConfig") then log("WARNING: Shared.GameConfig not readable; prices unknown, engine will try and observe.") end

        local roundOver, lastStatus = false, 0
        while token == myToken do
            if ctx.alive and not ctx.alive() then log("Host hub stopped. Engine stopping.") break end

            local blocked = ctx.blocked and ctx.blocked()
            if blocked then
                setStatus("Paused: " .. tostring(blocked))
                task.wait(1)
            else
                local done, why = matchEnded()
                if done then
                    if CONFIG.STOP_ON_END then
                        pcall(logRoundEnd, why)
                        log("Match ended (%s). Engine stopping.", tostring(why))
                        break
                    end
                    if not roundOver then
                        roundOver = true
                        pcall(logRoundEnd, why)
                        log("Round ended (%s). Waiting for the next round.", tostring(why))
                    end
                    setStatus("Round ended (" .. tostring(why) .. "), waiting for next round")
                    task.wait(1)
                else
                    if roundOver then roundOver = false resetRound() log("New round: state reset.") end
                    checkMap()
                    local paused, reason = matchPaused()
                    if paused then
                        setStatus("Waiting: " .. tostring(reason))
                        task.wait(0.5)
                    else
                        local ok, err = pcall(function()
                            if not schedulerStep(myToken) then waitForEvent(CONFIG.IDLE_WAIT, myToken) end
                        end)
                        if not ok then
                            log("LOOP ERROR: %s", tostring(err))
                            task.wait(1)
                        end
                    end
                end
            end

            if os.clock() - lastStatus >= 2 then
                lastStatus = os.clock()
                local _, mm = matchState()
                if not blocked and not matchEnded() and not matchPaused() then
                    setStatus(string.format("Farming | turn %s | wave %s | cash %s | dragons %d | last: %s%s",
                        turnDesc(), tostring(mm and mm:GetAttribute("Wave") or "?"),
                        tostring(getCash() or "?"), #getMyDragons(), E.Last, os.clock() < upgradeBlockedUntil and " | UPGRADES PAUSED" or ""))
                end
            end
            if os.clock() - (E._lastLogStatus or 0) >= CONFIG.STATUS_EVERY then
                E._lastLogStatus = os.clock()
                log("STATUS %s | mode=%s", E.StatusText, tostring(workingMode))
            end
        end

        if token == myToken then E.Running = false end
        flushLog(true)
        log("Engine stopped.")
        flushLog(true)
    end

    function E.Start()
        if E.Running then return true, "already running" end
        if game.PlaceId ~= CONFIG.EXPECTED_PLACE_ID then
            setStatus("Not in the match place (starts automatically there)")
            return false, "not in match place"
        end
        token = token + 1
        E.Running = true
        startedAt = os.clock()
        warnedEmptyList, warnedUngated = false, false
        local myToken = token
        task.spawn(function()
            local ok, err = pcall(loop, myToken)
            if not ok then
                log("ENGINE CRASH: %s", tostring(err))
                if token == myToken then E.Running = false end
                flushLog(true)
            end
        end)
        return true
    end

    function E.Stop()
        token = token + 1
        E.Running = false
        setStatus("Stopped")
    end

    return E
end

return createEngine

end)()
